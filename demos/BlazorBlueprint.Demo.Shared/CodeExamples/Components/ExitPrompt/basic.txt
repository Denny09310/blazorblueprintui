@* While the draft differs from what was saved, leaving is held and confirmed. *@
<BbExitPrompt Enabled="@dirty" />

<BbFormFieldInput Label="Draft title" Value="@title" ValueChanged="@OnTitleChanged" TValue="string" />
<BbButton OnClick="Save" Disabled="@(!dirty)">Save</BbButton>

@code {
    private const string Original = "Quarterly report";

    private string title = Original;
    private bool dirty;

    private void OnTitleChanged(string value)
    {
        title = value;
        dirty = !string.Equals(title, Original, StringComparison.Ordinal);
    }

    private void Save() => dirty = false;
}
