@using System.ComponentModel.DataAnnotations

<EditForm Model="validationModel" FormName="FormFieldFileUploadValidation">
    <DataAnnotationsValidator />
    <BbFormFieldFileUpload @bind-Files="validationModel.Files" Label="Supporting files" AriaLabel="Supporting files" />
    <BbButton Type="ButtonType.Submit" Class="mt-3">Validate selection</BbButton>
</EditForm>

@code {
    private readonly ValidationModel validationModel = new();

    private sealed class ValidationModel : IValidatableObject
    {
        public IReadOnlyList<FileUploadItem>? Files { get; set; }

        public IEnumerable<ValidationResult> Validate(ValidationContext validationContext)
        {
            if (Files is null || Files.Count == 0)
            {
                yield return new ValidationResult("Please make a selection.", [nameof(Files)]);
            }
        }
    }
}
