@using System.ComponentModel.DataAnnotations

<EditForm Model="validationModel" FormName="FormFieldDateRangePickerValidation">
    <DataAnnotationsValidator />
    <BbFormFieldDateRangePicker @bind-Value="validationModel.Value" Label="Travel dates" AriaLabel="Travel dates" />
    <BbButton Type="ButtonType.Submit" Class="mt-3">Validate selection</BbButton>
</EditForm>

@code {
    private readonly ValidationModel validationModel = new();

    private sealed class ValidationModel : IValidatableObject
    {
        public DateRange? Value { get; set; }

        public IEnumerable<ValidationResult> Validate(ValidationContext validationContext)
        {
            if (Value is null)
            {
                yield return new ValidationResult("Please make a selection.", [nameof(Value)]);
            }
        }
    }
}
