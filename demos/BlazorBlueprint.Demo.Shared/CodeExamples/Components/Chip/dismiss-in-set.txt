@* The set drops a dismissed value from the selection and reports it. *@
<BbChipSet TValue="string"
           SelectionMode="ChipSelectionMode.Multiple"
           Dismissible="true"
           @bind-Values="selected"
           OnDismiss="@(value => filters.Remove(value))">
    @foreach (var filter in filters)
    {
        <BbChip @key="filter" Value="@filter">@filter</BbChip>
    }
</BbChipSet>

@code {
    private readonly List<string> filters = ["Overdue", "Mine", "High priority"];
    private List<string> selected = ["Mine"];
}
